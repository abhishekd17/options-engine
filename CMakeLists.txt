cmake_minimum_required(VERSION 3.25)
project(
  options_engine
  VERSION 0.1.0
  DESCRIPTION "Options pricing, volatility surface and market-making engine"
  LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(OPTENG_BUILD_TESTS "Build unit tests" ON)
option(OPTENG_BUILD_BENCH "Build benchmarks" ON)
option(OPTENG_ENABLE_LTO "Enable link-time optimization for project targets" OFF)
option(OPTENG_NATIVE_ARCH "Tune project targets for the host CPU (-mcpu/-march=native)" OFF)
set(OPTENG_SANITIZER "none" CACHE STRING "Sanitizer set: none, asan (ASan+UBSan), tsan")
set_property(CACHE OPTENG_SANITIZER PROPERTY STRINGS none asan tsan)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(ProjectOptions)
include(Dependencies)
add_subdirectory(src)
add_subdirectory(tools)
if(OPTENG_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(OPTENG_BUILD_BENCH)
  add_subdirectory(bench)
endif()
